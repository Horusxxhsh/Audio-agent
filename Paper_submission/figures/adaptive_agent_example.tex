\begin{tikzpicture}[
    scale=1.0, transform shape,
    % Node styles
    block/.style={
        rectangle,
        rounded corners=3pt,
        minimum width=2.6cm,
        minimum height=0.8cm,

        drop shadow={shadow xshift=1pt, shadow yshift=-1pt, opacity=0.3}
    },
    perceptionblock/.style={
        block,
        fill=blue!12,

        text=black
    },
    memoryblock/.style={
        block,
        fill=cyan!12,
        draw=cyan!70,
        text=black
    },
    cognitionblock/.style={
        block,
        fill=purple!12,
        draw=purple!70,
        text=black
    },
    actionblock/.style={
        block,
        fill=green!10,
        draw=green!50!black!70,
        text=black
    },
    feedbackblock/.style={
        block,
        fill=red!8,
        draw=red!60,
        text=black
    },
    layerbox/.style={
        rectangle,
        rounded corners=5pt,
        draw=gray!50,
        fill=gray!5,
        inner sep=12pt,
        line width=0.7pt
    },
    layertitle/.style={
        font=\bfseries\small,
        text=blue!60!black,
        align=center
    },
    iobox/.style={
        rectangle,
        rounded corners=2pt,
        minimum width=1.8cm,
        minimum height=0.55cm,
        fill=white,
        draw=gray!60,
        font=\footnotesize,
        text=black
    },
    arrow/.style={
        -{Stealth[length=2.2mm,width=1.3mm]},
        thick,
        draw=gray!70
    },
    dataarrow/.style={
        -{Stealth[length=2.2mm,width=1.3mm]},
        thick,
        draw=blue!60
    },
    controlarrow/.style={
        -{Stealth[length=2.2mm,width=1.3mm]},
        thick,
        dashed,
        draw=red!60
    },
    memoryarrow/.style={
        -{Stealth[length=2.2mm,width=1.3mm]},
        thick,
        draw=cyan!60
    },
    label/.style={
        font=\scriptsize,
        text=gray!80,
        align=center
    }
]

% ============================================
% Layer 1: Perception Layer (Top)
% ============================================
\node[layerbox,minimum width=11cm,minimum height=2.6cm] (perceptionbox) at (0,6) {};
\node[layertitle,anchor=north west] at ($(perceptionbox.north west)+(0.3,-0.3)$) {Perception Layer};

% Text Intent Encoder
\node[perceptionblock] (textencoder) at (-3.2,6) {Text Intent Encoder};
\node[iobox,below=0.12cm of textencoder] (textinput) {Text Query $\mathbf{t}$};

% TRR Audio Encoder
\node[perceptionblock] (audioencoder) at (0,6) {TRR Audio Encoder};
\node[iobox,below=0.12cm of audioencoder] (audioinput) {Audio Query $\mathbf{a}$};

% Feature representations
\node[iobox,above=0.12cm of textencoder] (textfeat) {$\mathbf{f}_t$};
\node[iobox,above=0.12cm of audioencoder] (audiofeat) {$\mathbf{f}_a$};

% ============================================
% Layer 2: Memory Layer
% ============================================
\node[layerbox,minimum width=11cm,minimum height=2.6cm] (memorybox) at (0,2.8) {};
\node[layertitle,anchor=north west] at ($(memorybox.north west)+(0.3,-0.3)$) {Memory Layer};

% Semantic Memory
\node[memoryblock,minimum width=3.2cm] (semantic) at (-3.2,2.8) {Semantic Memory};
\node[label,below=0.03cm of semantic] {Static Knowledge Base};

% Episodic Memory (AEM)
\node[memoryblock,minimum width=3.2cm] (episodic) at (0,2.8) {Episodic Memory};
\node[label,below=0.03cm of episodic] {(AEM)};

% Memory outputs
\node[iobox,above=0.12cm of semantic] (semout) {$\mathbf{M}_s$};
\node[iobox,above=0.12cm of episodic] (epout) {$\mathbf{M}_e$};

% ============================================
% Layer 3: Cognition Layer
% ============================================
\node[layerbox,minimum width=11cm,minimum height=1.9cm] (cognitionbox) at (0,0.3) {};
\node[layertitle,anchor=north west] at ($(cognitionbox.north west)+(0.3,-0.3)$) {Cognition Layer};

% Adaptive Multimodal Fusion
\node[cognitionblock,minimum width=4.8cm] (fusion) at (-1.6,0.3) {Adaptive Multimodal Fusion};
\node[label,below=0.03cm of fusion] {$\mathbf{h} = \mathcal{F}(\mathbf{f}_t, \mathbf{f}_a, \mathbf{M}_s, \mathbf{M}_e)$};

% ============================================
% Layer 4: Action Layer (Bottom)
% ============================================
\node[layerbox,minimum width=11cm,minimum height=2.1cm] (actionbox) at (0,-2.3) {};
\node[layertitle,anchor=north west] at ($(actionbox.north west)+(0.3,-0.3)$) {Action Layer};

% Retrieval
\node[actionblock,minimum width=2.2cm] (retrieval) at (-3.5,-2) {Retrieval};

% Selection/Projection
\node[actionblock,minimum width=2.2cm] (selection) at (-1,-2) {Selection};
\node[label,below=0.03cm of selection] {\& Projection};

% Constraint Validation
\node[actionblock,minimum width=2.6cm] (validation) at (2,-2) {Validation};

% ============================================
% Output
% ============================================
\node[iobox,minimum width=3.2cm,fill=blue!8,draw=blue!60,thick,minimum height=0.65cm] (output) at (-0.5,-4) {DAW Plugin Parameters};
\node[label,below=0.03cm of output] {$\mathbf{p}^* = [p_1, p_2, \ldots, p_n]$};

% ============================================
% Learning Loop (Right side)
% ============================================
\node[feedbackblock,minimum width=2.4cm,minimum height=0.75cm] (feedback) at (4.2,2.8) {User Feedback};
\node[feedbackblock,minimum width=2.4cm,minimum height=0.75cm] (aemupdate) at (4.2,0.3) {AEM Update};

% ============================================
% Arrows - Data Flow
% ============================================
% Input to encoders
\draw[dataarrow] (textinput) -- (textencoder);
\draw[dataarrow] (audioinput) -- (audioencoder);

% Encoders to features
\draw[dataarrow] (textencoder) -- (textfeat);
\draw[dataarrow] (audioencoder) -- (audiofeat);

% Features to fusion
\draw[dataarrow] (textfeat.south) |- ($(fusion.north)+(0,0.28)$) -- (fusion.north);
\draw[dataarrow] (audiofeat.south) |- ($(fusion.north)+(0,0.28)$) -- (fusion.north);

% Memory to fusion
\draw[memoryarrow] (semantic) -- (semout);
\draw[memoryarrow] (episodic) -- (epout);
\draw[memoryarrow] (semout.north) |- ($(fusion.north)+(0,0.45)$) -- (fusion.north);
\draw[memoryarrow] (epout.north) |- ($(fusion.north)+(0,0.45)$) -- (fusion.north);

% Fusion to action
\draw[arrow] (fusion.south) -- ($(fusion.south)-(0,0.32)$) -| (retrieval.north);
\draw[arrow] (fusion.south) -- ($(fusion.south)-(0,0.32)$) -| (selection.north);
\draw[arrow] (fusion.south) -- ($(fusion.south)-(0,0.32)$) -| (validation.north);

% Action sequence
\draw[arrow] (retrieval) -- (selection);
\draw[arrow] (selection) -- (validation);
\draw[arrow] (validation.south) |- ($(output.north)+(0,0.28)$) -- (output.north);

% ============================================
% Arrows - Learning Loop
% ============================================
% Output to feedback
\draw[controlarrow] (output.east) -| ($(feedback.south)-(0,0.28)$) -- (feedback.south);
\node[label,anchor=west] at ($(feedback.east)+(0.08,0)$) {Rating/Adj.};

% Feedback to AEM update
\draw[controlarrow] (feedback) -- (aemupdate);
\node[label,anchor=west] at ($(aemupdate.east)+(0.08,0)$) {$\Delta \mathbf{M}_e$};

% AEM update to Episodic Memory
\draw[controlarrow] (aemupdate) -- (episodic);

% ============================================
% External Labels
% ============================================
% Multi-modal Input label (top)
\node[layertitle,font=\bfseries] at (0,7.3) {Multi-modal Input (Text + Audio)};
\draw[arrow,blue!50] (0,7.05) -- (0,6.75);

% Parameter Space connection
\draw[memoryarrow] (semantic.west) -| ($(semantic.west)+(-0.7,0)$) |- ($(semantic.south)-(0,0.35)$);
\node[label,anchor=east] at ($(semantic.west)+(-0.8,0)$) {$\mathcal{P}$};

% ============================================
% Legend

        \draw[controlarrow] (0,-0.35) -- (0.9,-0.35) node[right,font=\tiny] {Learning Loop};
    \end{tikzpicture}
};

\end{tikzpicture}

\caption{Architecture of the proposed Adaptive Parameter Synthesis Agent. The system processes multi-modal inputs (text and audio) through a perception layer, retrieves relevant knowledge from semantic and episodic memory, fuses information via adaptive multimodal fusion, and executes parameter synthesis through retrieval, selection, and validation. A learning loop enables continuous improvement through user feedback and AEM updates.}
\label{fig:adaptive_agent_architecture}
\end{figure*}
