    }

% ============================================
% Layer 1: Perception Layer (Top)
% ============================================
\node[layerbox,minimum width=12cm,minimum height=2.8cm] (perceptionbox) at (0,6.5) {};
\node[layertitle,anchor=north west] at ($(perceptionbox.north west)+(0.3,-0.3)$) {\textbf{Perception Layer}};

% Text Intent Encoder
\node[perceptionblock] (textencoder) at (-3.5,6.5) {Text Intent Encoder};
\node[iobox,below=0.15cm of textencoder] (textinput) {Text Query $\mathbf{t}$};

% TRR Audio Encoder
\node[perceptionblock] (audioencoder) at (0,6.5) {TRR Audio Encoder};
\node[iobox,below=0.15cm of audioencoder] (audioinput) {Audio Query $\mathbf{a}$};

% Feature representations
\node[iobox,above=0.15cm of textencoder] (textfeat) {$\mathbf{f}_t \in \mathbb{R}^{d}$};
\node[iobox,above=0.15cm of audioencoder] (audiofeat) {$\mathbf{f}_a \in \mathbb{R}^{d}$};

% ============================================
% Layer 2: Memory Layer
% ============================================
\node[layerbox,minimum width=12cm,minimum height=3.2cm] (memorybox) at (0,3.2) {};
\node[layertitle,anchor=north west] at ($(memorybox.north west)+(0.3,-0.3)$) {\textbf{Memory Layer}};

% Semantic Memory
\node[memoryblock,minimum width=3.5cm] (semantic) at (-3.5,3.2) {Semantic Memory};
\node[label,below=0.05cm of semantic] {Static Knowledge Base};
\node[label,below=0.35cm of semantic] {(Parameter Embeddings)};

% Episodic Memory (AEM)
\node[memoryblock,minimum width=3.5cm] (episodic) at (0,3.2) {Episodic Memory};
\node[label,below=0.05cm of episodic] {AEM (Adaptive Episodic};
\node[label,below=0.35cm of episodic] {Memory)};

% Memory outputs
\node[iobox,above=0.15cm of semantic] (semout) {$\mathbf{M}_s$};
\node[iobox,above=0.15cm of episodic] (epout) {$\mathbf{M}_e$};

% ============================================
% Layer 3: Cognition Layer
% ============================================
\node[layerbox,minimum width=12cm,minimum height=2.2cm] (cognitionbox) at (0,0.5) {};
\node[layertitle,anchor=north west] at ($(cognitionbox.north west)+(0.3,-0.3)$) {\textbf{Cognition Layer}};

% Adaptive Multimodal Fusion
\node[cognitionblock,minimum width=5cm] (fusion) at (-1.75,0.5) {Adaptive Multimodal Fusion};
\node[label,below=0.05cm of fusion] {$\mathbf{h} = \mathcal{F}(\mathbf{f}_t, \mathbf{f}_a, \mathbf{M}_s, \mathbf{M}_e)$};

% ============================================
% Layer 4: Action Layer (Bottom)
% ============================================
\node[layerbox,minimum width=12cm,minimum height=2.5cm] (actionbox) at (0,-2.5) {};
\node[layertitle,anchor=north west] at ($(actionbox.north west)+(0.3,-0.3)$) {\textbf{Action Layer}};

% Retrieval
\node[actionblock] (retrieval) at (-4,-2.2) {Retrieval};
\node[label,below=0.05cm of retrieval] {$\mathcal{R}(\mathbf{h}, \mathcal{P})$};

% Selection/Projection
\node[actionblock] (selection) at (-1,-2.2) {Selection};
\node[label,below=0.05cm of selection] {\& Projection};

% Constraint Validation
\node[actionblock] at (2,-2.2) (validation) {Constraint Validation};
\node[label,below=0.05cm of validation] {$\mathcal{C}(\mathbf{p}) \geq \theta$};

% ============================================
% Output
% ============================================
\node[iobox,minimum width=3.5cm,fill=ieeeblue!10,draw=ieeeblue!80,thick] (output) at (-1,-4.2) {DAW Plugin Parameters};
\node[label,below=0.05cm of output] {$\mathbf{p}^* = [p_1, p_2, \ldots, p_n]$};

% ============================================
% Learning Loop (Right side)
% ============================================
\node[feedbackblock,minimum width=2.5cm,minimum height=0.8cm] (feedback) at (4.5,3.2) {User Feedback};
\node[feedbackblock,minimum width=2.5cm,minimum height=0.8cm] (aemupdate) at (4.5,0.5) {AEM Update};

% ============================================
% Arrows - Data Flow
% ============================================
% Input to encoders
\draw[dataarrow] (textinput) -- (textencoder);
\draw[dataarrow] (audioinput) -- (audioencoder);

% Encoders to features
\draw[dataarrow] (textencoder) -- (textfeat);
\draw[dataarrow] (audioencoder) -- (audiofeat);

% Features to fusion
\draw[dataarrow] (textfeat) |- ($(fusion.north)+(0,0.3)$) -- (fusion);
\draw[dataarrow] (audiofeat) |- ($(fusion.north)+(0,0.3)$) -- (fusion);

% Memory to fusion
\draw[memoryarrow] (semantic) -- (semout);
\draw[memoryarrow] (episodic) -- (epout);
\draw[memoryarrow] (semout) |- ($(fusion.north)+(0,0.5)$) -- (fusion);
\draw[memoryarrow] (epout) |- ($(fusion.north)+(0,0.5)$) -- (fusion);

% Fusion to action
\draw[arrow] (fusion) -- ($(fusion.south)-(0,0.4)$) -| (retrieval);
\draw[arrow] (fusion) -- ($(fusion.south)-(0,0.4)$) -| (selection);
\draw[arrow] (fusion) -- ($(fusion.south)-(0,0.4)$) -| (validation);

% Action sequence
\draw[arrow] (retrieval) -- (selection);
\draw[arrow] (selection) -- (validation);
\draw[arrow] (validation) |- ($(output.north)+(0,0.3)$) -- (output);

% ============================================
% Arrows - Learning Loop
% ============================================
% Output to feedback (implicit user interaction)
\draw[controlarrow] (output.east) -| ($(feedback.south)-(0,0.3)$) -- (feedback);
\node[label,anchor=west] at ($(feedback.east)+(0.1,0)$) {Rating/Adjustment};

% Feedback to AEM update
\draw[controlarrow] (feedback) -- (aemupdate);
\node[label,anchor=west] at ($(aemupdate.east)+(0.1,0)$) {$\Delta \mathbf{M}_e$};

% AEM update to Episodic Memory
\draw[controlarrow] (aemupdate) -- (episodic);

% ============================================
% External Labels
% ============================================

\node[layertitle,font=\bfseries] at (0,8.2) {Multi-modal Input};
\draw[arrow,ieeeblue!60] (0,7.9) -- (0,7.5);

% Parameter Space connection
\draw[memoryarrow] (semantic.west) -| ($(semantic.west)+(-0.8,0)$) |- ($(semantic.south)-(0,0.5)$);
\node[label,anchor=east] at ($(semantic.west)+(-0.9,0)$) {$\mathcal{P}$};
