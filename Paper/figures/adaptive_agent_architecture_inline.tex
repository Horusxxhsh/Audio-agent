\begin{tikzpicture}[
    scale=0.9, transform shape,
    % Node styles
    block/.style={
        rectangle,
        rounded corners=2.5pt,
        minimum width=2.4cm,
        minimum height=0.75cm,
        text centered,
        font=\footnotesize\bfseries,
        drop shadow={shadow xshift=0.8pt, shadow yshift=-0.8pt, opacity=0.25}
    },
    perceptionblock/.style={
        block,
        fill=blue!12,
        draw=blue!70,
        text=black
    },
    memoryblock/.style={
        block,
        fill=cyan!12,
        draw=cyan!70,
        text=black
    },
    cognitionblock/.style={
        block,
        fill=purple!12,
        draw=purple!70,
        text=black
    },
    actionblock/.style={
        block,
        fill=green!10,
        draw=green!50!black!70,
        text=black
    },
    feedbackblock/.style={
        block,
        fill=red!8,
        draw=red!60,
        text=black
    },
    layerbox/.style={
        rectangle,
        rounded corners=4pt,
        draw=gray!50,
        fill=gray!5,
        inner sep=10pt,
        line width=0.6pt
    },
    layertitle/.style={
        font=\bfseries\footnotesize,
        text=blue!60!black,
        align=center
    },
    iobox/.style={
        rectangle,
        rounded corners=2pt,
        minimum width=1.6cm,
        minimum height=0.5cm,
        fill=white,
        draw=gray!60,
        font=\scriptsize,
        text=black
    },
    arrow/.style={
        -{Stealth[length=2mm,width=1.2mm]},
        thick,
        draw=gray!70
    },
    dataarrow/.style={
        -{Stealth[length=2mm,width=1.2mm]},
        thick,
        draw=blue!60
    },
    controlarrow/.style={
        -{Stealth[length=2mm,width=1.2mm]},
        thick,
        dashed,
        draw=red!60
    },
    memoryarrow/.style={
        -{Stealth[length=2mm,width=1.2mm]},
        thick,
        draw=cyan!60
    },
    label/.style={
        font=\tiny,
        text=gray!80,
        align=center
    }
]

% ============================================
% Title
% ============================================
\node[font=\bfseries] at (0,8) {Adaptive Parameter Synthesis Agent Architecture};

% ============================================
% Layer 1: Perception Layer (Top)
% ============================================
\node[layerbox,minimum width=10.5cm,minimum height=2.4cm] (perceptionbox) at (0,6.2) {};
\node[layertitle,anchor=north west] at ($(perceptionbox.north west)+(0.25,-0.25)$) {Perception Layer};

% Text Intent Encoder
\node[perceptionblock] (textencoder) at (-3,6.2) {Text Intent Encoder};
\node[iobox,below=0.1cm of textencoder] (textinput) {Text Query $\mathbf{t}$};

% TRR Audio Encoder
\node[perceptionblock] (audioencoder) at (0,6.2) {TRR Audio Encoder};
\node[iobox,below=0.1cm of audioencoder] (audioinput) {Audio Query $\mathbf{a}$};

% Feature representations
\node[iobox,above=0.1cm of textencoder] (textfeat) {$\mathbf{f}_t$};
\node[iobox,above=0.1cm of audioencoder] (audiofeat) {$\mathbf{f}_a$};

% ============================================
% Layer 2: Memory Layer
% ============================================
\node[layerbox,minimum width=10.5cm,minimum height=2.8cm] (memorybox) at (0,3) {};
\node[layertitle,anchor=north west] at ($(memorybox.north west)+(0.25,-0.25)$) {Memory Layer};

% Semantic Memory
\node[memoryblock,minimum width=3cm] (semantic) at (-3,3) {Semantic Memory};
\node[label,below=0.02cm of semantic] {Static Knowledge};

% Episodic Memory (AEM)
\node[memoryblock,minimum width=3cm] (episodic) at (0,3) {Episodic Memory};
\node[label,below=0.02cm of episodic] {(AEM)};

% Memory outputs
\node[iobox,above=0.1cm of semantic] (semout) {$\mathbf{M}_s$};
\node[iobox,above=0.1cm of episodic] (epout) {$\mathbf{M}_e$};

% ============================================
% Layer 3: Cognition Layer
% ============================================
\node[layerbox,minimum width=10.5cm,minimum height=1.8cm] (cognitionbox) at (0,0.4) {};
\node[layertitle,anchor=north west] at ($(cognitionbox.north west)+(0.25,-0.25)$) {Cognition Layer};

% Adaptive Multimodal Fusion
\node[cognitionblock,minimum width=4.5cm] (fusion) at (-1.5,0.4) {Adaptive Multimodal Fusion};
\node[label,below=0.02cm of fusion] {$\mathbf{h} = \mathcal{F}(\mathbf{f}_t, \mathbf{f}_a, \mathbf{M}_s, \mathbf{M}_e)$};

% ============================================
% Layer 4: Action Layer (Bottom)
% ============================================
\node[layerbox,minimum width=10.5cm,minimum height=2.2cm] (actionbox) at (0,-2.2) {};
\node[layertitle,anchor=north west] at ($(actionbox.north west)+(0.25,-0.25)$) {Action Layer};

% Retrieval
\node[actionblock,minimum width=2cm] (retrieval) at (-3.5,-1.9) {Retrieval};

% Selection/Projection
\node[actionblock,minimum width=2cm] (selection) at (-1,-1.9) {Selection};

% Constraint Validation
\node[actionblock,minimum width=2.4cm] (validation) at (1.8,-1.9) {Validation};

% ============================================
% Output
% ============================================
\node[iobox,minimum width=3cm,fill=blue!8,draw=blue!60,thick,minimum height=0.6cm] (output) at (-0.5,-3.8) {DAW Plugin Parameters};
\node[label,below=0.02cm of output] {$\mathbf{p}^* = [p_1, \ldots, p_n]$};

% ============================================
% Learning Loop (Right side)
% ============================================
\node[feedbackblock,minimum width=2.2cm,minimum height=0.7cm] (feedback) at (4,3) {User Feedback};
\node[feedbackblock,minimum width=2.2cm,minimum height=0.7cm] (aemupdate) at (4,0.4) {AEM Update};

% ============================================
% Arrows - Data Flow
% ============================================
% Input to encoders
\draw[dataarrow] (textinput) -- (textencoder);
\draw[dataarrow] (audioinput) -- (audioencoder);

% Encoders to features
\draw[dataarrow] (textencoder) -- (textfeat);
\draw[dataarrow] (audioencoder) -- (audiofeat);

% Features to fusion
\draw[dataarrow] (textfeat.south) |- ($(fusion.north)+(0,0.25)$) -- (fusion.north);
\draw[dataarrow] (audiofeat.south) |- ($(fusion.north)+(0,0.25)$) -- (fusion.north);

% Memory to fusion
\draw[memoryarrow] (semantic) -- (semout);
\draw[memoryarrow] (episodic) -- (epout);
\draw[memoryarrow] (semout.north) |- ($(fusion.north)+(0,0.4)$) -- (fusion.north);
\draw[memoryarrow] (epout.north) |- ($(fusion.north)+(0,0.4)$) -- (fusion.north);

% Fusion to action
\draw[arrow] (fusion.south) -- ($(fusion.south)-(0,0.35)$) -| (retrieval.north);
\draw[arrow] (fusion.south) -- ($(fusion.south)-(0,0.35)$) -| (selection.north);
\draw[arrow] (fusion.south) -- ($(fusion.south)-(0,0.35)$) -| (validation.north);

% Action sequence
\draw[arrow] (retrieval) -- (selection);
\draw[arrow] (selection) -- (validation);
\draw[arrow] (validation.south) |- ($(output.north)+(0,0.25)$) -- (output.north);

% ============================================
% Arrows - Learning Loop
% ============================================
% Output to feedback
\draw[controlarrow] (output.east) -| ($(feedback.south)-(0,0.25)$) -- (feedback.south);
\node[label,anchor=west] at ($(feedback.east)+(0.05,0)$) {Rating};

% Feedback to AEM update
\draw[controlarrow] (feedback) -- (aemupdate);
\node[label,anchor=west] at ($(aemupdate.east)+(0.05,0)$) {$\Delta \mathbf{M}_e$};

% AEM update to Episodic Memory
\draw[controlarrow] (aemupdate) -- (episodic);

% ============================================
% External Labels
% ============================================
% Multi-modal Input label (top)
\node[layertitle] at (0,7.5) {Multi-modal Input};
\draw[arrow,blue!50] (0,7.3) -- (0,6.95);

% Parameter Space
\draw[memoryarrow] (semantic.west) -| ($(semantic.west)+(-0.6,0)$) |- ($(semantic.south)-(0,0.4)$);
\node[label,anchor=east] at ($(semantic.west)+(-0.7,0)$) {$\mathcal{P}$};

% ============================================
% Legend
% ============================================
\node[anchor=south east,inner sep=3pt] at (5.5,-4.5) {
    \begin{tikzpicture}[scale=0.7]
        \draw[dataarrow] (0,0) -- (0.8,0) node[right,font=\tiny] {Data};
        \draw[controlarrow] (0,-0.3) -- (0.8,-0.3) node[right,font=\tiny] {Feedback};
    \end{tikzpicture}
};

\end{tikzpicture}
